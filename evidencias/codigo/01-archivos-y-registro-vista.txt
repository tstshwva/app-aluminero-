using System.Security.Cryptography;
using System.Text;
using MySql.Data.MySqlClient;

namespace Aluformix.Api;

public static class AuthEndpoints
{
    public static void MapAuth(this WebApplication app)
    {
        var auth=app.MapGroup("/api/auth").RequireRateLimiting("auth");
        auth.MapPost("/register",async(RegisterRequest request,AuthStore store,CancellationToken ct)=>
        {
            var email=AuthStore.Email(request.Email);var name=request.Name?.Trim();var phone=request.Phone?.Trim();
            if(email is null || name is not {Length: > 0 and <=120} || !AuthStore.PasswordValid(request.Password) || !AuthStore.PhoneValid(phone))
                return Results.BadRequest(new {error="Indica nombre, correo, teléfono válidos y contraseña de 12 a 128 caracteres."});
            try {var user=await store.CreateAsync(name,email,request.Password!,3,phone,ct);return Results.Created("/api/auth/me",user);}
            catch(MySqlException ex) when(ex.Number==1062) {return Results.Conflict(new {error="No fue posible registrar la cuenta con esos datos."});}
        });
        auth.MapPost("/login",async(LoginRequest request,AuthStore store,HttpContext context,CancellationToken ct)=>
        {
            context.Response.Headers.CacheControl="no-store";
            var session=await store.LoginAsync(request,ct);
            return session is null?Results.Json(new {error="No fue posible iniciar sesión. Revisa tus datos o inténtalo más tarde."},statusCode:401):Results.Ok(session);
        });
        auth.MapGet("/me",(HttpContext context)=>
        {
            context.Response.Headers.CacheControl="no-store";
            return Results.Ok(new Account(ulong.Parse(context.User.FindFirstValue(ClaimTypes.NameIdentifier)!),
                context.User.Identity!.Name!,context.User.FindFirstValue(ClaimTypes.Email)!,context.User.FindFirstValue(ClaimTypes.Role)!));
        }).RequireAuthorization();
        auth.MapPost("/logout",async(HttpContext context,AuthStore store,CancellationToken ct)=>
        {
            await using var db=store.Connection();await db.OpenAsync(ct);
            await using var cmd=new MySqlCommand("UPDATE auth_sessions SET revoked_at=UTC_TIMESTAMP(6) WHERE token_hash=@hash",db);
            cmd.Parameters.AddWithValue("@hash",context.User.FindFirstValue("session_hash"));await cmd.ExecuteNonQueryAsync(ct);
            return Results.NoContent();
        }).RequireAuthorization();
        auth.MapPost("/staff",async(StaffRequest request,AuthStore store,CancellationToken ct)=>
        {
            var email=AuthStore.Email(request.Email);var name=request.Name?.Trim();
            if(email is null || name is not {Length: >0 and <=120} || !AuthStore.PasswordValid(request.Password)
                || request.Role is not ("Administrador" or "Operador")) return Results.BadRequest(new {error="Datos de cuenta inválidos."});
            try {var user=await store.CreateAsync(name,email,request.Password!,request.Role=="Administrador"?1:2,null,ct);return Results.Ok(user);}
            catch(MySqlException ex) when(ex.Number==1062) {return Results.Conflict(new {error="No fue posible crear la cuenta."});}
        }).RequireAuthorization("Admin");

        // Solo habilitado por configuración externa explícita, y únicamente si no existe administrador.
        auth.MapPost("/setup-admin",async(LoginRequest request,HttpContext context,AuthStore store,IConfiguration config,CancellationToken ct)=>
        {
            var secret=config["Setup:AdminKey"];
            var supplied=context.Request.Headers["X-Setup-Key"].ToString();
            if(secret is not {Length: >=32} || !CryptographicOperations.FixedTimeEquals(SHA256.HashData(Encoding.UTF8.GetBytes(secret)),SHA256.HashData(Encoding.UTF8.GetBytes(supplied))))
                return Results.NotFound();

            var name=config["Setup:AdminName"]?.Trim();
            if(email is null || name is not {Length: >0 and <=120} || !AuthStore.PasswordValid(request.Password)) return Results.BadRequest();

        });
